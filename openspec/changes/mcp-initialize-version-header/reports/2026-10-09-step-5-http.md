# Step 5: real HTTP smoke

From the isolated built worktree, executed `npx -y -p node@24 node ../tasks/mcp-r2-smoke.mjs`: exit 0,
**11 curl checks passed**. The full harness below is preserved for reproduction. Save it outside the application,
then execute it with the built worktree as cwd. It starts its own loopback Next server on 3248 with a synthetic token,
uses curl.exe, stops its child server and deletes its own temporary payloads in finally. Ports 3240 and 3246 are untouched.

The first run passed ten cases but the bearer case incorrectly sent both valid and invalid Authorization headers,
receiving 200. The harness was corrected to send only the wrong bearer; the full rerun passed. This was a harness
error, not a product fix or a claim of an authentication bypass.

The timeout uploads a valid 24,000-byte body at 1,024 bytes/second; HTTP 408 arrived in 10,064 ms. The streaming
boundary uses chunked transfer without Content-Length. Unit tests separately cover an understated Content-Length,
which cannot faithfully travel through normal HTTP framing as a full request.

## Reproducer

```javascript
import assert from 'node:assert/strict';
import { spawn } from 'node:child_process';
import { mkdtemp, writeFile, rm } from 'node:fs/promises';
import { tmpdir } from 'node:os';
import { join } from 'node:path';
import { once } from 'node:events';

const port = 3248;
const root = await mkdtemp(join(tmpdir(), 'cited-mcp-smoke-'));
const server = spawn(process.execPath, ['node_modules/next/dist/bin/next', 'start', '--hostname', '127.0.0.1', '--port', String(port)], {
  cwd: process.cwd(), env: { ...process.env, CITED_MCP_TOKEN: 'smoke-only-token', DATABASE_URL: join(root, 'unused.sqlite') },
  stdio: ['ignore', 'pipe', 'pipe'], windowsHide: true,
});
let serverLog = '';
server.stdout.on('data', (data) => { serverLog += data; });
server.stderr.on('data', (data) => { serverLog += data; });
const url = `http://127.0.0.1:${port}/api/mcp`;
const init = { jsonrpc: '2.0', id: 1, method: 'initialize', params: { protocolVersion: '2025-11-25' } };
const cases = [
  ['initialize-new-header', JSON.stringify(init), 200, []],
  ['initialize-notification', JSON.stringify({ jsonrpc: '2.0', method: 'initialize' }), 400, []],
  ['tools-list-new-header', JSON.stringify({ jsonrpc: '2.0', id: 2, method: 'tools/list' }), 400, []],
  ['tools-call-new-header', JSON.stringify({ jsonrpc: '2.0', id: 3, method: 'tools/call', params: { name: 'cited_search', arguments: { query: 'sample' } } }), 400, []],
  ['malformed-json', '{', 400, []],
  ['exact-limit', JSON.stringify(init).padEnd(1048576), 200, []],
  ['declared-overflow', JSON.stringify(init).padEnd(1048577), 413, []],
  ['chunked-overflow', JSON.stringify(init).padEnd(1048577), 413, ['-H', 'Transfer-Encoding: chunked']],
  ['total-deadline', JSON.stringify(init).padEnd(24000), 408, ['--limit-rate', '1024']],
  ['origin-guard', JSON.stringify(init), 403, ['-H', 'Origin: https://foreign.example']],
  ['bearer-guard', JSON.stringify(init), 401, ['-H', 'Authorization: Bearer wrong']],
];
try {
  for (let index = 0; index < 100; index++) {
    if (server.exitCode !== null) throw new Error(serverLog);
    try { await fetch(url); break; } catch { await new Promise((resolve) => setTimeout(resolve, 100)); }
  }
  for (const [name, body, expected, extra] of cases) {
    const file = join(root, `${name}.json`);
    await writeFile(file, body);
    const args = ['--silent', '--show-error', '--max-time', '20', '--write-out', '\nHTTP:%{http_code}',
      '-H', 'Content-Type: application/json', ...(name === 'bearer-guard' ? [] : ['-H', 'Authorization: Bearer smoke-only-token']),
      '-H', 'MCP-Protocol-Version: 2025-11-25', '--data-binary', `@${file}`, ...extra, url];
    const started = performance.now();
    const curl = spawn('curl.exe', args, { windowsHide: true });
    let output = '', errors = '';
    curl.stdout.on('data', (data) => { output += data; });
    curl.stderr.on('data', (data) => { errors += data; });
    const [exit] = await once(curl, 'close');
    const status = Number(output.match(/HTTP:(\d+)/)?.[1]);
    console.log(JSON.stringify({ name, expected, status, exit, elapsedMs: Math.round(performance.now() - started), output, errors }));
    assert.equal(status, expected, name);
    assert.equal(exit, 0, name);
    if (expected === 200) assert.equal(JSON.parse(output.split('\nHTTP:')[0]).result.protocolVersion, '2025-06-18');
  }
  console.log('PASS: 11 curl checks against isolated Next build');
} finally {
  server.kill();
  if (server.exitCode === null) await once(server, 'exit');
  await rm(root, { recursive: true, force: true });
}

```
